# Create a network for the containers to communicate
docker network create miasoftnanus-network
# Command in order to Up Sonarqube
docker compose -f ./sonarqube/docker-compose.yaml up
docker compose -f ./sonarqube/docker-compose.yaml up -d

# Command in order to down Sonarqube
docker compose -f ./sonarqube/docker-compose.yaml down

# Command in order to Up Jenkins
docker compose -f ./jenkins/docker-compose.yaml up
docker compose -f ./jenkins/docker-compose.yaml up -d

# Command in order to down Jenkins
docker compose -f ./jenkins/docker-compose.yaml down

# Test Jenkins can reach SonarQube
docker exec jenkins curl -I http://sonarqube:9000

# Configure SonarQube in Jenkins
## In Jenkins, use the container name as hostname when configuring the SonarQube server:
http://sonarqube:9000

# Command in order to Up MSSQL Server
docker compose -f ./mssql-server/docker-compose.yaml up
docker compose -f ./mssql-server/docker-compose.yaml up -d
# Command in order to down MSSQL Server
docker compose -f ./mssql-server/docker-compose.yaml down

# command in order to check logs and status of the containers
docker compose -f ./sonarqube/docker-compose.yaml logs -f
docker compose -f ./jenkins/docker-compose.yaml logs -f
docker compose -f ./mssql-server/docker-compose.yaml logs -f

# command in order to check the status of the containers
docker compose -f ./sonarqube/docker-compose.yaml ps
docker compose -f ./jenkins/docker-compose.yaml ps
docker compose -f ./mssql-server/docker-compose.yaml ps

docker compose -f ./sonarqube/docker-compose.yaml status
docker compose -f ./jenkins/docker-compose.yaml status
docker compose -f ./mssql-server/docker-compose.yaml status

docker compose -f ./sonarqube/docker-compose.yaml stop
docker compose -f ./sonarqube/docker-compose.yaml start
docker compose -f ./jenkins/docker-compose.yaml stop
docker compose -f ./jenkins/docker-compose.yaml start
docker compose -f ./mssql-server/docker-compose.yaml stop
docker compose -f ./mssql-server/docker-compose.yaml start

# command in order to Up Jenkins Agents
docker compose -f jenkins/jenkins-agent/docker-compose-3-agents.yaml --env-file jenkins/jenkins-agent/.env up -d
docker compose -f jenkins/jenkins-agent/docker-compose-3-agents.yaml stop
docker compose -f jenkins/jenkins-agent/docker-compose-3-agents.yaml start
docker compose -f jenkins/jenkins-agent/docker-compose-3-agents.yaml down
